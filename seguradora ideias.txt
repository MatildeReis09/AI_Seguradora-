# Definimos a importância de cada variável no risco final
pesos_finais = np.array([0.15, 0.15, 0.40, 0.10, 0.20]) # Age, Smoker, Gravidade, Profit, Sinistralidade
df_trabalho['MEU_RISK_SCORE'] = (X_scaled @ pesos_finais) * 100

# --- IMPLEMENTAÇÃO DA INTELIGÊNCIA ARTIFICIAL ---

# 7. Método do Cotovelo (Para justificar o K=5 no relatório)
inercias = []
for k in range(1, 10):
    model_elbow = KMeans(n_clusters=k, random_state=42, n_init=10)
    model_elbow.fit(X_scaled)
    inercias.append(model_elbow.inertia_)

plt.figure(figsize=(8, 4))
plt.plot(range(1, 10), inercias, marker='o', color='red')
plt.title('Método do Cotovelo (Elbow Method)')
plt.xlabel('Número de Grupos (K)')
plt.ylabel('Inércia')
plt.savefig('grafico_cotovelo.png') # Salva para o teu trabalho

# 8. Executar o K-Means final com K=5
kmeans = KMeans(n_clusters=5, random_state=42, n_init=10)
clusters = kmeans.fit_predict(X_scaled)

# 9. Adicionar o resultado ao dataset e ordenar o Score (1 a 5)
df_original['Calculo_risk_score'] = clusters

# Organizar para que 5 seja o maior risco (baseado em hospitalizações)
ordem = df_original.groupby('Calculo_risk_score')['hospitalizations_last_3yrs'].mean().sort_values().index
mapeamento = {antigo: novo + 1 for novo, antigo in enumerate(ordem)}
df_original['Calculo_risk_score'] = df_original['Calculo_risk_score'].map(mapeamento)
